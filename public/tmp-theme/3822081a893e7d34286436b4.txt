(function(){
  'use strict';
  if(window.__PMY_PDP_CHECKOUT_ENHANCE__)return;

  var loading=null;

  function loadScript(src,id){
    return new Promise(function(resolve,reject){
      var existing=document.getElementById(id);
      if(existing){
        if(existing.dataset.loaded==='1')return resolve();
        existing.addEventListener('load',resolve,{once:true});
        existing.addEventListener('error',reject,{once:true});
        return;
      }
      var script=document.createElement('script');
      script.id=id;
      script.src=src;
      script.async=false;
      script.addEventListener('load',function(){
        script.dataset.loaded='1';
        resolve();
      },{once:true});
      script.addEventListener('error',reject,{once:true});
      document.head.appendChild(script);
    });
  }

  function enhance(){
    if(loading)return loading;
    loading=loadScript({{ 'countries-data.js' | asset_url | json }},'pmy-pdp-countries')

(function(){
  function resetCheckoutLayer(){
    var drawer=document.getElementById('pmy-checkout-drawer');
    var backdrop=document.querySelector('[data-pmy-checkout-backdrop]');
    var trigger=document.querySelector('[data-pmy-checkout-trigger]');
    if(drawer && drawer.getAttribute('aria-hidden')!=='false'){
      drawer.classList.remove('is-open');
    }
    if(backdrop && !(drawer && drawer.classList.contains('is-open'))){
      backdrop.classList.remove('is-open');
      backdrop.setAttribute('aria-hidden','true');
    }
    if(trigger && !(drawer && drawer.classList.contains('is-open'))){
      trigger.classList.remove('is-open');
      trigger.setAttribute('aria-expanded','false');
    }
    if(!(drawer && drawer.classList.contains('is-open'))){
      document.body.classList.remove('pmy-checkout-lock');
    }
  }
  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded',resetCheckoutLayer,{once:true});
  }else{
    resetCheckoutLayer();
  }
  document.addEventListener('shopify:section:load',function(){
    window.setTimeout(resetCheckoutLayer,0);
  });
})();
</script>
